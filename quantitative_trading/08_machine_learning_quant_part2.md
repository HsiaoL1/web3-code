```python
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras.models import Sequential, Model
from tensorflow.keras.layers import Dense, LSTM, GRU, Conv1D, MaxPooling1D, Dropout, Attention, Input, MultiHeadAttention
from tensorflow.keras.optimizers import Adam
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau
from sklearn.preprocessing import MinMaxScaler
import matplotlib.pyplot as plt

class FinancialNeuralNetwork:
    """
    金融神经网络基类
    """
    
    def __init__(self, sequence_length: int = 60, features_dim: int = 10):
        self.sequence_length = sequence_length
        self.features_dim = features_dim
        self.model = None
        self.scaler = MinMaxScaler()
        self.history = None
        
    def prepare_sequences(self, data: pd.DataFrame, target_col: str = None) -> tuple:
        """
        准备序列数据
        """
        
        # 数据标准化
        scaled_data = self.scaler.fit_transform(data)
        
        X, y = [], []
        
        for i in range(self.sequence_length, len(scaled_data)):
            # 特征序列
            X.append(scaled_data[i-self.sequence_length:i])
            
            # 目标值
            if target_col is not None:
                target_idx = data.columns.get_loc(target_col)
                y.append(scaled_data[i, target_idx])
            else:
                # 默认预测下一个时间点的所有特征
                y.append(scaled_data[i])
        
        return np.array(X), np.array(y)
    
    def create_mlp_model(self, hidden_layers: list = [128, 64, 32], 
                        dropout_rate: float = 0.2, output_dim: int = 1):
        """
        创建多层感知机模型
        """
        
        model = Sequential()
        
        # 输入层
        model.add(Dense(hidden_layers[0], 
                       input_shape=(self.sequence_length * self.features_dim,),
                       activation='relu'))
        model.add(Dropout(dropout_rate))
        
        # 隐藏层
        for units in hidden_layers[1:]:
            model.add(Dense(units, activation='relu'))
            model.add(Dropout(dropout_rate))
        
        # 输出层
        if output_dim == 1:
            model.add(Dense(1, activation='linear'))  # 回归
        else:
            model.add(Dense(output_dim, activation='softmax'))  # 分类
        
        return model
    
    def create_lstm_model(self, lstm_units: list = [50, 50], 
                         dropout_rate: float = 0.2, output_dim: int = 1):
        """
        创建LSTM模型
        """
        
        model = Sequential()
        
        # LSTM层
        for i, units in enumerate(lstm_units):
            return_sequences = i < len(lstm_units) - 1
            
            if i == 0:
                model.add(LSTM(units, 
                              return_sequences=return_sequences,
                              input_shape=(self.sequence_length, self.features_dim)))
            else:
                model.add(LSTM(units, return_sequences=return_sequences))
            
            model.add(Dropout(dropout_rate))
        
        # 全连接层
        model.add(Dense(32, activation='relu'))
        model.add(Dropout(dropout_rate))
        
        # 输出层
        if output_dim == 1:
            model.add(Dense(1, activation='linear'))
        else:
            model.add(Dense(output_dim, activation='softmax'))
        
        return model
    
    def create_cnn_lstm_model(self, cnn_filters: int = 64, cnn_kernel: int = 3,
                             lstm_units: int = 50, output_dim: int = 1):
        """
        创建CNN-LSTM混合模型
        """
        
        model = Sequential()
        
        # CNN层
        model.add(Conv1D(filters=cnn_filters, 
                        kernel_size=cnn_kernel,
                        activation='relu',
                        input_shape=(self.sequence_length, self.features_dim)))
        model.add(MaxPooling1D(pool_size=2))
        model.add(Dropout(0.2))
        
        # 再添加一层CNN
        model.add(Conv1D(filters=cnn_filters//2, kernel_size=cnn_kernel, activation='relu'))
        model.add(Dropout(0.2))
        
        # LSTM层
        model.add(LSTM(lstm_units, return_sequences=False))
        model.add(Dropout(0.2))
        
        # 全连接层
        model.add(Dense(32, activation='relu'))
        model.add(Dropout(0.2))
        
        # 输出层
        if output_dim == 1:
            model.add(Dense(1, activation='linear'))
        else:
            model.add(Dense(output_dim, activation='softmax'))
        
        return model
    
    def train_model(self, X_train, y_train, X_val, y_val, 
                   model_type: str = 'lstm', epochs: int = 100, 
                   batch_size: int = 32, **kwargs):
        """
        训练模型
        """
        
        # 创建模型
        if model_type == 'mlp':
            # 重塑输入用于MLP
            X_train_reshaped = X_train.reshape(X_train.shape[0], -1)
            X_val_reshaped = X_val.reshape(X_val.shape[0], -1)
            self.model = self.create_mlp_model(**kwargs)
            X_train_input, X_val_input = X_train_reshaped, X_val_reshaped
        elif model_type == 'lstm':
            self.model = self.create_lstm_model(**kwargs)
            X_train_input, X_val_input = X_train, X_val
        elif model_type == 'cnn_lstm':
            self.model = self.create_cnn_lstm_model(**kwargs)
            X_train_input, X_val_input = X_train, X_val
        
        # 编译模型
        self.model.compile(optimizer=Adam(learning_rate=0.001),
                          loss='mse' if kwargs.get('output_dim', 1) == 1 else 'sparse_categorical_crossentropy',
                          metrics=['mae'] if kwargs.get('output_dim', 1) == 1 else ['accuracy'])
        
        # 回调函数
        callbacks = [
            EarlyStopping(patience=10, restore_best_weights=True),
            ReduceLROnPlateau(factor=0.5, patience=5, min_lr=1e-7)
        ]
        
        # 训练
        self.history = self.model.fit(
            X_train_input, y_train,
            validation_data=(X_val_input, y_val),
            epochs=epochs,
            batch_size=batch_size,
            callbacks=callbacks,
            verbose=1
        )
        
        return self.history
    
    def predict(self, X_test):
        """
        预测
        """
        if self.model is None:
            raise ValueError("Model not trained yet")
        
        return self.model.predict(X_test)
    
    def plot_training_history(self):
        """
        绘制训练历史
        """
        if self.history is None:
            return
        
        fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 4))
        
        # 损失
        ax1.plot(self.history.history['loss'], label='Training Loss')
        ax1.plot(self.history.history['val_loss'], label='Validation Loss')
        ax1.set_title('Model Loss')
        ax1.set_xlabel('Epoch')
        ax1.set_ylabel('Loss')
        ax1.legend()
        
        # 指标
        metric_key = 'mae' if 'mae' in self.history.history else 'accuracy'
        ax2.plot(self.history.history[metric_key], label=f'Training {metric_key.upper()}')
        ax2.plot(self.history.history[f'val_{metric_key}'], label=f'Validation {metric_key.upper()}')
        ax2.set_title(f'Model {metric_key.upper()}')
        ax2.set_xlabel('Epoch')
        ax2.set_ylabel(metric_key.upper())
        ax2.legend()
        
        plt.tight_layout()
        plt.show()

class TransformerModel:
    """
    Transformer模型用于金融时间序列
    """
    
    def __init__(self, seq_length: int, d_model: int = 128, num_heads: int = 8, 
                 num_layers: int = 4, dff: int = 512):
        self.seq_length = seq_length
        self.d_model = d_model
        self.num_heads = num_heads
        self.num_layers = num_layers
        self.dff = dff
        self.model = None
        
    def create_transformer_model(self, input_dim: int, output_dim: int = 1):
        """
        创建Transformer模型
        """
        
        # 输入
        inputs = Input(shape=(self.seq_length, input_dim))
        
        # 位置编码
        x = self.positional_encoding(inputs)
        
        # Transformer层
        for _ in range(self.num_layers):
            x = self.transformer_layer(x)
        
        # 全局平均池化
        x = tf.keras.layers.GlobalAveragePooling1D()(x)
        
        # 输出层
        if output_dim == 1:
            outputs = Dense(1, activation='linear')(x)
        else:
            outputs = Dense(output_dim, activation='softmax')(x)
        
        model = Model(inputs=inputs, outputs=outputs)
        return model
    
    def positional_encoding(self, x):
        """
        位置编码
        """
        # 简化的位置编码实现
        seq_len = tf.shape(x)[1]
        pos = tf.range(seq_len, dtype=tf.float32)[:, tf.newaxis]
        
        # 线性投影到d_model维度
        x = Dense(self.d_model)(x)
        
        # 添加位置信息
        pos_encoding = tf.sin(pos / 10000**(2 * tf.range(self.d_model, dtype=tf.float32) / self.d_model))
        pos_encoding = pos_encoding[:, :self.d_model]
        
        return x + pos_encoding[tf.newaxis, :, :]
    
    def transformer_layer(self, x):
        """
        Transformer层
        """
        # 多头注意力
        attention_output = MultiHeadAttention(
            num_heads=self.num_heads, 
            key_dim=self.d_model // self.num_heads
        )(x, x)
        
        # 残差连接和层归一化
        x = tf.keras.layers.LayerNormalization()(x + attention_output)
        
        # 前馈网络
        ffn_output = tf.keras.layers.Dense(self.dff, activation='relu')(x)
        ffn_output = tf.keras.layers.Dense(self.d_model)(ffn_output)
        
        # 残差连接和层归一化
        x = tf.keras.layers.LayerNormalization()(x + ffn_output)
        
        return x
```
